    <script>
        // 🔐 **1. تعديل الباسورد هنا:**
        const correctPassword = "123"; // غيره لأي باسورد تحبه (تاريخ خطوبة، عيد ميلادها، الخ)

        function checkPassword() {
            const inputVal = document.getElementById('pass-input').value;
            const errorMsg = document.getElementById('error-msg');
            
            if(inputVal === correctPassword) {
                document.getElementById('password-screen').style.display = 'none';
                document.getElementById('main-content').style.display = 'block';
                
                // تشغيل الأغنية
                const song = document.getElementById('bday-song');
                song.play().catch(e => console.log("Autoplay blocked:", e));
            } else {
                errorMsg.style.display = 'block';
            }
        }

        // 📸 **2. صور الذكريات (روابط الصور بتاعتكم):**
        const photos = [
            "https://images.unsplash.com/photo-1518199266791-5375a83190b7?w=500&auto=format&fit=crop&q=60", // صورة 1
            "https://images.unsplash.com/photo-1516589178581-6cd7833ae3b2?w=500&auto=format&fit=crop&q=60", // صورة 2
            "https://images.unsplash.com/photo-1494774157365-9e04c6720e47?w=500&auto=format&fit=crop&q=60"  // صورة 3
        ];
        let currentPhotoIdx = 0;

        function nextPhoto() {
            currentPhotoIdx = (currentPhotoIdx + 1) % photos.length;
            document.getElementById('current-photo').src = photos[currentPhotoIdx];
        }

        // ⏳ **3. حساب العدادات:**
        // تاريخ بداية العلاقة (عدل السنة والشهر واليوم)
        const relStartDate = new Date("2024-01-01T00:00:00");
        // تاريخ عيد الميلاد / الخطوبة القادم (5 نوفمبر)
        const targetDate = new Date("2026-11-05T00:00:00"); 

        function updateCounters() {
            const now = new Date();

            // عداد العلاقة
            const diffRel = now - relStartDate;
            const daysRel = Math.floor(diffRel / (1000 * 60 * 60 * 24));
            document.getElementById('rel-counter').innerText = `${daysRel} يوم من الحب والسعادة ❤️`;

            // عداد عيد الميلاد والخطوبة
            const diffTarget = targetDate - now;
            if(diffTarget > 0) {
                const daysTarget = Math.floor(diffTarget / (1000 * 60 * 60 * 24));
                const hoursTarget = Math.floor((diffTarget % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60));
                document.getElementById('bd-counter').innerText = `${daysTarget} يوم و ${hoursTarget} ساعة بس! 🥳`;
            } else {
                document.getElementById('bd-counter').innerText = "النهارده أحلى عيد ميلاد وأحلى خطوبة! 🎉💍";
            }
        }
        setInterval(updateCounters, 1000);
        updateCounters();

        // 💬 **4. الأسئلة التفاعلية:**
        function answerQuiz(choice) {
            const resultDiv = document.getElementById('quiz-result');
            if(choice === 1) {
                resultDiv.innerHTML = "إجابة صحيحة بنسبة 100%، ما شاء الله عليك فاهمها صح 😂❤️";
            } else {
                resultDiv.innerHTML = "محاولة خادعة! طبعاً هي اللي بتحب أكتر بكتير 😜❤️";
            }
        }
    </script>
